def build_rdd_approve_payload(backend_refs, comment=""):
    sr_id = backend_refs.get("sr_id", "")
    create_payload = backend_refs.get("create_payload") or {}
    inner = create_payload.get("payload") or {}

    # Direct backend_refs IDs take precedence over create_payload

        "user_action": None,

        "tenant_profile_id": tenant_profile_id,  # ← direct refs override
        "property_id": property_id,
        ...
        "status": "APPROVED",
    }
